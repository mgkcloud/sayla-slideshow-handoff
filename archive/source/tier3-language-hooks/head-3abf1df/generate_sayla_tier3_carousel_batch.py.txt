def run_higgsfield(outdir, slide):
    prompt = f"{IDENTITY}\n\nScene: {slide['scene']}\n\nStyle: {STYLE}\n\n{NEG}"
    p = outdir/f"prompt_{slide['role']}.txt"
    p.write_text(prompt)
    log = outdir/f"higgsfield_{slide['role']}.log"
    cmd = ['higgsfield','generate','create','nano_banana_2','--image',str(REFS[0]),'--image',str(REFS[1]),'--aspect_ratio','9:16','--resolution','2k','--wait','--wait-timeout','20m']
    print('GEN', outdir.name, slide['role'], flush=True)
    with open(p,'rb') as stdin:
        proc = subprocess.run(cmd, stdin=stdin, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, timeout=1500)
    log.write_text(proc.stdout)
    urls = re.findall(r'https?://\S+', proc.stdout)
    if proc.returncode != 0 or not urls:
        raise RuntimeError(proc.stdout[-1200:])
    url=urls[-1]
    rawdir=outdir/'raw'; rawdir.mkdir(exist_ok=True)
    raw=rawdir/f"{slide['role']}.png"
    req=urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
    raw.write_bytes(urllib.request.urlopen(req, timeout=180).read())
    return url, raw, p

def cover(im):
    im=im.convert('RGB')
    iw,ih=im.size
    s=max(W/iw,H/ih)
    im=im.resize((int(iw*s),int(ih*s)),Image.LANCZOS)
    return im.crop(((im.width-W)//2,(im.height-H)//2,(im.width-W)//2+W,(im.height-H)//2+H))

def wrap(text,font,max_w):
    d=ImageDraw.Draw(Image.new('RGB',(10,10)))
    words=text.lower().split(); lines=[]; cur=''
    for w in words:
        test=(cur+' '+w).strip()
        if d.textbbox((0,0),test,font=font)[2] <= max_w:
            cur=test
        else:
            if cur: lines.append(cur)
            cur=w
    if cur: lines.append(cur)
    return lines

def overlay(raw, text, idx):
    im=cover(Image.open(raw))
    im=ImageEnhance.Contrast(im).enhance(1.06)
    im=ImageEnhance.Color(im).enhance(1.04).convert('RGBA')
    shade=Image.new('RGBA',(W,H),(0,0,0,0)); sd=ImageDraw.Draw(shade)
    for y in range(H):
        a=0
        if y<740: a=max(a,int(66*(1-y/740)))
        if y>1260: a=max(a,int(42*((y-1260)/(H-1260))))
        if a: sd.line((0,y,W,y), fill=(0,0,0,a))
    im.alpha_composite(shade)
    y = 230 if idx < 3 else 1120
    size=84
    while size>=46:
        f=ImageFont.truetype(str(FONT_BOLD), size)
        lines=wrap(text,f,900)
        lh=int(size*1.06); total=lh*len(lines)
        if total<440: break
        size-=4
    halo=Image.new('RGBA',(W,H),(0,0,0,0)); hd=ImageDraw.Draw(halo)
    cy=y+total//2
    for r,a in [(420,18),(310,28),(220,40)]:
        hd.ellipse((W//2-r,cy-r,W//2+r,cy+r), fill=(0,0,0,a))
    im.alpha_composite(halo)
    d=ImageDraw.Draw(im)
    yy=y
    for line in lines:
        bbox=d.textbbox((0,0),line,font=f,stroke_width=7)
        x=(W-(bbox[2]-bbox[0]))//2
        d.text((x,yy),line,font=f,fill=(255,255,255,255),stroke_width=7,stroke_fill=(0,0,0,245))
        yy+=lh
    return im.convert('RGB')

def preview(outdir, finals):
    TW,TH=270,480
    sheet=Image.new('RGB',(TW*3+44,TH+78),(10,10,12)); d=ImageDraw.Draw(sheet)
    f=ImageFont.truetype(str(FONT_REG),24)
    for i,p in enumerate(finals,1):
        img=Image.open(p).resize((TW,TH),Image.LANCZOS)
        x=10+(i-1)*(TW+12)
        sheet.paste(img,(x,10)); d.text((x+8,TH+25),f'slide {i}',font=f,fill=(235,230,225))
    out=outdir/'carousel_preview.jpg'; sheet.save(out,quality=92); return out
