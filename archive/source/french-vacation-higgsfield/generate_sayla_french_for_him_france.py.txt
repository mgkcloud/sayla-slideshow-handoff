def run_higgsfield(slide):
    prompt = f"{IDENTITY}\n\nScene: {slide['scene']}\n\nStyle: {STYLE}\n\n{NEG}"
    prompt_path = OUT/f"prompt_{slide['id']}.txt"
    prompt_path.write_text(prompt)
    log_path = OUT/f"higgsfield_{slide['id']}.log"
    cmd = [
        'higgsfield','generate','create','nano_banana_2',
        '--image',str(REFS[0]),'--image',str(REFS[1]),
        '--aspect_ratio','9:16','--resolution','2k',
        '--wait','--wait-timeout','20m'
    ]
    print(f"=== generating {slide['id']} ===", flush=True)
    with open(prompt_path,'rb') as stdin, open(log_path,'w') as log:
        proc = subprocess.run(cmd, stdin=stdin, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, timeout=1500)
        log.write(proc.stdout)
    urls = re.findall(r'https?://\S+', proc.stdout)
    if proc.returncode != 0 or not urls:
        raise RuntimeError(f"higgsfield failed for {slide['id']}: {proc.stdout[-1000:]}")
    url = urls[-1].strip()
    raw = RAW/f"{slide['id']}.png"
    req = urllib.request.Request(url, headers={'User-Agent':'Mozilla/5.0'})
    raw.write_bytes(urllib.request.urlopen(req, timeout=180).read())
    return url, raw, prompt

def fit_cover(im):
    im = im.convert('RGB')
    iw,ih = im.size
    s = max(W/iw,H/ih)
    im = im.resize((int(iw*s), int(ih*s)), Image.LANCZOS)
    return im.crop(((im.width-W)//2, (im.height-H)//2, (im.width-W)//2+W, (im.height-H)//2+H))

def wrap(text, font, max_w):
    d = ImageDraw.Draw(Image.new('RGB',(10,10)))
    words = text.lower().split(); lines=[]; cur=''
    for w in words:
        t=(cur+' '+w).strip()
        if d.textbbox((0,0),t,font=font,stroke_width=0)[2] <= max_w:
            cur=t
        else:
            if cur: lines.append(cur)
            cur=w
    if cur: lines.append(cur)
    return lines

def overlay(im, text, y):
    im = fit_cover(im)
    im = ImageEnhance.Contrast(im).enhance(1.06)
    im = ImageEnhance.Color(im).enhance(1.04).convert('RGBA')
    # top/bottom vignette only; no card/box
    shade = Image.new('RGBA',(W,H),(0,0,0,0)); sd=ImageDraw.Draw(shade)
    for yy in range(H):
        a=0
        if yy < 690: a=max(a, int(58*(1-yy/690)))
        if yy > 1240: a=max(a, int(42*((yy-1240)/(H-1240))))
        if a: sd.line((0,yy,W,yy), fill=(0,0,0,a))
    im.alpha_composite(shade)
    d=ImageDraw.Draw(im)
    size=78
    while size >= 46:
        font=ImageFont.truetype(str(FONT_BOLD), size)
        lines=wrap(text,font,900)
        lh=int(size*1.08); total=lh*len(lines)
        if total < 430: break
        size -= 4
    # soft behind text, still not a card
    halo=Image.new('RGBA',(W,H),(0,0,0,0)); hd=ImageDraw.Draw(halo)
    cy=y+total//2
    for r,a in [(410,18),(310,26),(220,36)]:
        hd.ellipse((W//2-r,cy-r,W//2+r,cy+r), fill=(0,0,0,a))
    im.alpha_composite(halo)
    d=ImageDraw.Draw(im)
    yy=y
    for line in lines:
        bbox=d.textbbox((0,0), line, font=font, stroke_width=7)
        x=(W-(bbox[2]-bbox[0]))//2
        d.text((x,yy), line, font=font, fill=(255,255,255,255), stroke_width=7, stroke_fill=(0,0,0,245))
        yy += lh
    return im.convert('RGB')

def contact_sheet(paths):
    TW,TH=216,384
    sheet=Image.new('RGB',(TW*5+60,TH+74),(10,10,12)); d=ImageDraw.Draw(sheet)
    f=ImageFont.truetype(str(FONT_REG),24)
    for i,p in enumerate(paths,1):
        img=Image.open(p).resize((TW,TH),Image.LANCZOS)
        x=10+(i-1)*(TW+10)
        sheet.paste(img,(x,10)); d.text((x+8,TH+25),f'slide {i}',font=f,fill=(235,230,225))
    out=OUT/'carousel_preview.jpg'; sheet.save(out,quality=92); return out
