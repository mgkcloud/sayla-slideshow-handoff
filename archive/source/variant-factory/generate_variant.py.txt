def clone_scene_image(scene, out_path):
    """Generate a fresh Sophia image staged for `scene` via FAL nano-banana edit with face refs."""
    import base64, io, requests
    # Owner credential helper omitted
    urls = []
    for p in sorted((BASE / 'character' / 'references').glob('ref_0*.png'))[:3]:
        im = Image.open(p).convert('RGB')
        im.thumbnail((768, 768))
        buf = io.BytesIO()
        im.save(buf, 'JPEG', quality=88)
        urls.append('data:image/jpeg;base64,' + base64.b64encode(buf.getvalue()).decode())
    prompt = (f"Put this exact woman ({SOPHIA_FACE}) in a new photo: {scene}. {CLONE_STYLE}")
    r = requests.post('https://fal.run/fal-ai/nano-banana/edit',
                      headers={'Authorization': '<API_CREDENTIAL_REFERENCE>'},
                      json={'prompt': prompt, 'image_urls': urls, 'num_images': 1, 'aspect_ratio': '9:16'},
                      timeout=300)
    r.raise_for_status()
    img = requests.get(r.json()['images'][0]['url'], timeout=120)
    img.raise_for_status()
    out_path.write_bytes(img.content)
    return out_path


def validate_variants(items, n):
    if not isinstance(items, list) or not items:
        return False
    return all(isinstance(v, dict) and v.get('overlay') and v.get('caption') and v.get('variant_name')
               for v in items[:n])
