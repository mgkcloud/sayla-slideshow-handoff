"""Normalise every slide to the 1080x1920 TikTok carousel frame.

Two shapes go in. The generated lifestyle slides are 1520x2688, which is 9:16 to
within half a percent, so they just resize. The deterministic phone screenshots
are 1179x2556 (9:19.5) and cannot be cropped to fit: the lock screen keeps its
notifications near the bottom and its clock near the top, so trimming either end
throws away the thing the slide exists to show. They are fitted by height onto a
blurred, darkened enlargement of themselves, which is what the platforms do to
an over-tall image anyway.
"""

from PIL import Image, ImageFilter

import tiktok_text

OUT = "/tmp/tt/gen/deck"
CREATIVE = "<original-host-path>"

# Keeping them named rather than numbered means a deck folder says what it is.

DEFAULT_PAIR = ("car", "street")


def slides(hook, second):
    return [
        ("1_real", VARIANTS[hook], "fill"),
        ("2_imessage", f"{CREATIVE}/a_2_imessage.png", "fit"),
        ("3_real", VARIANTS[second], "fill"),
        ("4_lock", f"{CREATIVE}/a_2_lock_real.png", "fit"),
        ("5_payoff", f"{CREATIVE}/a_3_payoff.png", "fit"),
    ]

# Copy lives here, not in the image-generation prompt. Text baked in by the model
# cannot be corrected or split-tested, and comes out in the wrong typeface; drawn
# here it is a variable and one run rebuilds the deck. Slides 2, 4 and 5 carry
# their own text already, so they are absent from this table on purpose.
CAPTIONS = {
    "1_real": ("my grandma calls me in Albanian\n"
               "and i answer in English\n"
               "every single time", 0.50, 0.66),
    "3_real": ("every app taught me Tirana Albanian\n"
               "my family speaks Gheg\n"
               "it is not the same language", 0.50, 0.66),
}


def fill(im):
    """Cover the frame, centre-cropping the sliver that overhangs."""
    s = max(W / im.width, H / im.height)
    im = im.resize((round(im.width * s), round(im.height * s)), Image.LANCZOS)
    x, y = (im.width - W) // 2, (im.height - H) // 2
    return im.crop((x, y, x + W, y + H))


def fit(im, margin=30):
    """Fit by height over a blurred enlargement of the same image."""
    bg = fill(im).filter(ImageFilter.GaussianBlur(38)).point(lambda v: int(v * 0.55))
    inner = H - margin * 2
    s = inner / im.height
    fg = im.resize((round(im.width * s), inner), Image.LANCZOS)
    bg.paste(fg, ((W - fg.width) // 2, margin))
    return bg


def build(hook, second, out_dir):
    os.makedirs(out_dir, exist_ok=True)
    deck = slides(hook, second)
    for name, path, mode in deck:
        im = Image.open(path).convert("RGB")
        out = (fill if mode == "fill" else fit)(im)
        if name in CAPTIONS:
            text, cx, top = CAPTIONS[name]
            out = tiktok_text.draw(out, text, cx=cx, top=top)
        out.save(f"{out_dir}/{name}.png")
        print(f"{name:12s} {im.size} -> {out.size}  ({mode})"
              f"{'  + caption' if name in CAPTIONS else ''}")

    sheet = Image.new("RGB", (216 * len(deck), 384), (18, 18, 18))
    for i, (name, _, _) in enumerate(deck):
        sheet.paste(Image.open(f"{out_dir}/{name}.png").resize((216, 384), Image.LANCZOS), (i * 216, 0))
    sheet.save(f"{out_dir}/_sheet.png")
    print(f"wrote {out_dir}/_sheet.png")
    return f"{out_dir}/_sheet.png"


if __name__ == "__main__":
    import sys
    hook, second = (sys.argv[1:3] + list(DEFAULT_PAIR))[:2]
    build(hook, second, f"{OUT}_{hook}_{second}")
