<script>
function render(cfg) {
  document.getElementById('wall').style.backgroundImage = cfg.wallpaper ? `url("${cfg.wallpaper}")` : '';
  if (cfg.wallpaperColor) document.getElementById('wall').style.backgroundColor = cfg.wallpaperColor;
  document.getElementById('scrim').style.background = `rgba(0,0,0,${cfg.scrim ?? 0.16})`;
  document.getElementById('date').textContent = cfg.date;
  document.getElementById('time').textContent = cfg.time;
  document.getElementById('sbtime').textContent = cfg.statusTime || cfg.time;

  const nb = document.getElementById('notifs');
  if (cfg.notifyTop != null) { nb.style.top = cfg.notifyTop + 'px'; nb.style.bottom = 'auto'; }
  else { nb.style.bottom = (cfg.notifyBottom ?? 92) + 'px'; nb.style.top = 'auto'; }
  if (cfg.hideChrome) {
    for (const id of ['torch','camera','indicator']) document.getElementById(id).style.display = 'none';
  }
  const r = document.documentElement.style;
  for (const [k,v] of Object.entries(cfg.vars || {})) r.setProperty('--'+k, v);

  nb.innerHTML = '';
  for (const n of cfg.notifications) {
    const card = document.createElement('div');
    card.className = 'card';

    const icon = document.createElement('div');
    icon.className = 'icon';
    if (n.iconImage) icon.style.backgroundImage = `url("${n.iconImage}")`;
    else { icon.style.background = n.iconColor || '#666'; icon.textContent = n.iconGlyph || ''; }

    const content = document.createElement('div');
    content.className = 'content';
    const head = document.createElement('div'); head.className = 'head';
    const t = document.createElement('div'); t.className = 'title'; t.textContent = n.title;
    const w = document.createElement('div'); w.className = 'when';  w.textContent = n.when || '';
    head.appendChild(t); head.appendChild(w);
    const b = document.createElement('div'); b.className = 'body'; b.textContent = n.body;
    content.appendChild(head); content.appendChild(b);

    card.appendChild(icon); card.appendChild(content);
    nb.appendChild(card);
  }
  document.body.dataset.ready = '1';
}
</script>
