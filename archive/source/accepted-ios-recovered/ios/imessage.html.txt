<script>
function render(cfg) {
  document.body.className = cfg.theme || 'dark';

  document.getElementById('sbtime').textContent = cfg.statusTime || '9:41';
  document.getElementById('unread').textContent = cfg.unread ? String(cfg.unread) : '';
  document.querySelector('#peername span').textContent = cfg.peer || '';

  const av = document.getElementById('avatar');
  if (cfg.peerImage) av.style.backgroundImage = `url("${cfg.peerImage}")`;
  else av.textContent = (cfg.peerInitials != null ? cfg.peerInitials
        : (cfg.peer || '').split(/\s+/).map(w => w[0] || '').slice(0, 2).join('').toUpperCase());

  const r = document.documentElement.style;
  for (const [k, v] of Object.entries(cfg.vars || {})) r.setProperty('--' + k, v);

  const t = document.getElementById('thread');
  t.innerHTML = '';
  const msgs = cfg.messages || [];
  msgs.forEach((m, i) => {
    if (m.stamp) {
      const s = document.createElement('div');
      s.className = 'stamp';
      s.innerHTML = m.stamp.replace(/^(\S+)/, '<b>$1</b>');
      t.appendChild(s);
      return;
    }
    const inc = !m.out;
    const prev = msgs[i - 1];
    const next = msgs[i + 1];
    const first = !prev || prev.stamp || !!prev.out !== !!m.out;
    const last = !next || next.stamp || !!next.out !== !!m.out;

    const row = document.createElement('div');
    row.className = 'row ' + (inc ? 'inc' : 'out') + (first ? ' first' : '') + (m.sms ? ' sms' : '');
    const b = document.createElement('div');
    b.className = 'bubble' + (last ? ' tail' : '');
    b.textContent = m.text;
    row.appendChild(b);
    t.appendChild(row);

    if (m.receipt) {
      const rc = document.createElement('div');
      rc.style.cssText = 'text-align:right;font-size:11px;color:#8E8E93;margin-top:3px;padding-right:3px';
      rc.innerHTML = m.receipt.replace(/^(\S+)/, '<b style="font-weight:600">$1</b>');
      t.appendChild(rc);
    }
  });

  document.body.dataset.ready = '1';
}
</script>
