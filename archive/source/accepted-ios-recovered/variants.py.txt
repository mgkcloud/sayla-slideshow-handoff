# The four new concepts want a specific still, not any still: she has just read
# the thread the viewer is about to see. a/b/c could sit on any of the ten
# scenes because their hooks were statements about her life, so a supermarket
# aisle worked as well as a bedroom. These are the ten seconds after a message
# landed, and a scene where she is cheerfully talking to her phone contradicts
# the caption over her face.
#
# The pairing is data rather than a note in a commit, because build_post.py's
# `scene` argument and the variant key are chosen together or the post is
# incoherent, and nothing else in the tree records that.
SCENE_FOR = {k: v["scene"] for k, v in VARIANTS.items() if "scene" in v}


def render_ios(key, slide, cfg):
    out = os.path.join(OUT, f"{key}_{slide}.png")
    tmp = os.path.join(OUT, f".{key}_{slide}.json")
    with open(tmp, "w") as fh:
        json.dump(cfg, fh)
    subprocess.run([sys.executable, os.path.join(IOS, "render.py"), tmp, out], check=True)
    os.remove(tmp)


def render_lock(key, v):
    cfg = dict(CHASSIS)
    cfg.update(v["lock"])
    cfg["screen"] = "lock"
    wall = os.path.join(HERE, "walls", v["wallpaper"])
    if os.path.exists(wall):
        cfg["wallpaper"] = wall
    render_ios(key, "2_lock", cfg)


def render_imessage(key, v):
    cfg = dict(v["thread"])
    cfg["screen"] = "imessage"
    cfg.setdefault("theme", "dark")
    render_ios(key, "2_imessage", cfg)


def find_photo(name):
    """Resolve a slide-1 photo by name, from faces/ or from out/.

    faces/ holds the three selfies generated by hand in August. out/ is where
    ads/generate.py drops everything it fetches. Looking in both is the whole
    reconciliation: until this existed the ads pipeline could build a prompt,
    fire it and download the result, and there was still no way to put that
    result on a slide — render_hook only ever opened faces/.

    An absolute or relative path is taken as given, so a one-off render does not
    need the file moved into either directory first.
    """
    if os.path.sep in name:
        return name if os.path.exists(name) else None
    for d in (os.path.join(HERE, "faces"), OUT):
        p = os.path.join(d, name)
        if os.path.exists(p):
            return p
    return None


def render_hook(key, v, photo=None, suffix=""):
    face = find_photo(photo or v["face"])
    if not face:
        print("skip hook, no photo:", photo or v["face"])
        return
    out = os.path.join(OUT, f"{key}_1_hook{suffix}.png")
    subprocess.run([sys.executable, os.path.join(HERE, "slides.py"),
                    "hook", out, v["hook"], "--photo", face,
                    "--y", str(v.get("y", 0.60)), "--size", str(v.get("size", 68))],
                   check=True)
    return out


APP_SHOT = os.path.join(
    os.path.dirname(HERE), "Code", "ShqipTutor",
    "appstore-screenshots", "raw", "08-live-tutor-14s.png")


def render_payoff(key, v):
    """Slide 3: the real app, so the reveal has something to land on."""
    if not os.path.exists(APP_SHOT):
        print("skip payoff, no app screenshot")
        return
    out = os.path.join(OUT, f"{key}_3_payoff.png")
    subprocess.run([sys.executable, os.path.join(HERE, "slides.py"),
                    "hook", out, v["payoff"], "--photo", APP_SHOT,
                    "--y", "0.75", "--size", "60"], check=True)


if __name__ == "__main__":
    os.makedirs(OUT, exist_ok=True)
    args = sys.argv[1:]

    # `hook a s2_arcade.png` renders slide 1 only, against a named photo, to
    # <key>_1_hook_<photo stem>.png. Testing a slide-1 prompt means rendering
    # the same line over several photos and looking at them side by side, and
    # the full run rebuilds three iOS screens per variant through headless
    # Chrome to get there.
    if args[:1] == ["hook"]:
        key, photos = args[1], args[2:]
        for photo in photos or [VARIANTS[key]["face"]]:
            stem = os.path.splitext(os.path.basename(photo))[0]
            print(render_hook(key, VARIANTS[key], photo, f"_{stem}"))
        sys.exit()

    for k in args or list(VARIANTS):
        v = VARIANTS[k]
        render_hook(k, v)
        render_lock(k, v)
        render_imessage(k, v)
        render_payoff(k, v)
